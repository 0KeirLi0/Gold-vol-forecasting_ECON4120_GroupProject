% #1 destination, #2 section number, #3 label
% current = gold fill; earlier = gold wash; later = grey
\newcommand{\navchip}[3]{%
  \hyperlink{#1}{%
    \ifnum\value{section}=#2
      \colorbox{goldaccent}{\makebox[0.168\paperwidth][c]{\textcolor{white}{\scriptsize\bfseries #3}}}%
    \else
      \ifnum\value{section}>#2
        \colorbox{goldaccent!25}{\makebox[0.168\paperwidth][c]{\textcolor{goldaccent}{\scriptsize\bfseries #3}}}%
      \else
        \colorbox{black!8}{\makebox[0.168\paperwidth][c]{\textcolor{black!40}{\scriptsize #3}}}%
      \fi
    \fi
  }%
}
